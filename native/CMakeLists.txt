cmake_minimum_required(VERSION 3.16)
project(arb_rt CXX)
set(CMAKE_CXX_STANDARD 17)
# NVIDIA's lib is built with the static runtime (/MT)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
# Both folders are fetched by BUILD-NATIVE.bat (they are not shipped with the game).
set(VKH "${CMAKE_CURRENT_SOURCE_DIR}/third_party/Vulkan-Headers/include")
set(DLSS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/DLSS")
add_executable(arb-rt src/main_rt.cpp src/probe.cpp src/vk_min.cpp src/glvk.cpp src/replay.cpp src/serve.cpp)
if(WIN32)
  target_link_libraries(arb-rt PRIVATE ws2_32)
endif()
target_include_directories(arb-rt PRIVATE src ${VKH})
if(EXISTS "${DLSS}/include/nvsdk_ngx.h" AND NOT ARB_NO_DLSS)
  message(STATUS "DLSS SDK found: building with DLSS")
  target_compile_definitions(arb-rt PRIVATE ARB_DLSS)
  target_include_directories(arb-rt PRIVATE "${DLSS}/include")
  target_link_libraries(arb-rt PRIVATE "${DLSS}/lib/Windows_x86_64/x64/nvsdk_ngx_s.lib")
  # NVIDIA's runtime DLL must sit next to the exe (release build of it).
  add_custom_command(TARGET arb-rt POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${DLSS}/lib/Windows_x86_64/rel/nvngx_dlss.dll" "$<TARGET_FILE_DIR:arb-rt>/nvngx_dlss.dll")
else()
  message(STATUS "Building WITHOUT DLSS (SDK not found)")
endif()
